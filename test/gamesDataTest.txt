	function toggleMenu(isOpen) {
		const sideMenu = document.getElementById('navMenu');
		const firstMenuItem = sideMenu.querySelector('a');
		const toggleButton = document.getElementById('desktopHeaderToggleA');

		sideMenu.setAttribute('aria-hidden', !isOpen);
		toggleButton.setAttribute('aria-expanded', isOpen);

		if (isOpen) {
			// Make the menu items focusable
			sideMenu.querySelectorAll('a').forEach(item => {
				item.setAttribute('tabindex', '0');
			});
			// Move focus to the first menu item
			firstMenuItem.focus();

			var navContainer = document.getElementById('js-screen-content');
			var focusableElements = navContainer.querySelectorAll('a[href], button');
			var lastFocusableElement = focusableElements[focusableElements.length - 1];

			$(lastFocusableElement).on('focusout', function() {
				setTimeout(function() {
					if (!sideMenu.contains(document.activeElement)) {
						closeNav();
					}
				}, 0);
			});

		} else {
			// Remove focus from menu items
			sideMenu.querySelectorAll('a').forEach(item => {
				item.setAttribute('tabindex', '-1');
			});
			// Move focus back to the toggle button
			toggleButton.focus();
		}
	}

	$("#desktopHeaderToggleA").click(function(e) {
		e.preventDefault();

		e.stopPropagation();
	});

		e.preventDefault();
		toggleMenu(false);
		e.stopPropagation();

	const RGBToHSL = (colorHex) => {
		//console.log(colorHex);
		if (colorHex.includes("rgb")) {
			colorHex = rgbToHex(colorHex);
		}

		// console.log(colorHex);
		let r = parseInt(colorHex.substr(0, 2), 16);
		let g = parseInt(colorHex.substr(2, 2), 16);
		let b = parseInt(colorHex.substr(4, 2), 16);
		r /= 255;
		g /= 255;
		b /= 255;
		const l = Math.max(r, g, b);
		const s = l - Math.min(r, g, b);
		const h = s
			? l === r
				? (g - b) / s
				: l === g
					? 2 + (b - r) / s
					: 4 + (r - g) / s
			: 0;
		return {
			h: 60 * h < 0 ? 60 * h + 360 : 60 * h,
			s: 100 * (s ? (l <= 0.5 ? s / (2 * l - s) : s / (2 - (2 * l - s))) : 0),
			l: (100 * (2 * l - s)) / 2,
		};
	};

	const rgbToHex = (rgb) => {
		//pull out the rgb values
		var rgbValues = rgb.split("(")[1].split(")")[0];

		// split it into separate numbers:

		rgbValues = rgbValues.split(",");
		// Convert the single numbers to hex

		var convertedToHex = rgbValues.map(function (x) {             //For each array element
		x = parseInt(x).toString(16);      //Convert to a base16 string
		return (x.length == 1) ? "0" + x : x;  //Add zero if we get only one character
		})
		//And glue it back together:

		convertedToHex = convertedToHex.join("");
		return convertedToHex;
    }

	const getEventPodsForDay = (filteredEvents, dayStart) => {
		//events are already filtered by day
		const dayEnd = new Date(dayStart.getTime());
		dayEnd.setDate(dayStart.getDate() + 1);
		const theDaysEvents = filteredEvents.filter(eventItem => {
            return (new Date(`${eventItem.start} 00:00:00`) >= dayStart) && (new Date(`${eventItem.start} 00:00:00`) < dayEnd);
        });
		if (theDaysEvents && theDaysEvents.length > 0) {
			let eventHtml = "";
			theDaysEvents.forEach((event) => {
				eventHtml += getEventPod(event);
			})
			return eventHtml;
		} else {
			return "";
		}
	}

	const getEventPodsForHours = (filteredEvents, hours) => {

			return "";
		}
